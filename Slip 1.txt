Q1 Write a Java program to print the sum of elements of the array. Also 
display array elements in ascending order.

import java.util.*;

class ArraySort
{
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);

        int a[] = new int[5];
        int sum = 0, temp;

        System.out.println("Enter 5 elements:");

        for(int i=0; i<5; i++)
        {
            a[i] = sc.nextInt();
            sum = sum + a[i];
        }

        // Sorting in ascending order
        for(int i=0; i<5; i++)
        {
            for(int j=i+1; j<5; j++)
            {
                if(a[i] > a[j])
                {
                    temp = a[i];
                    a[i] = a[j];
                    a[j] = temp;
                }
            }
        }

        System.out.println("Sum = " + sum);

        System.out.println("Ascending Order:");
        for(int i=0; i<5; i++)
        {
            System.out.print(a[i] + " ");
        }
    }
}
