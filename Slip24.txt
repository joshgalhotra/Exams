Q1) Define a class MyNumber having one private integer data member. Write 
a default constructor initialize it to 0 and another constructor to initialize 
it to a value. Write methods isNegative, isPositive, isOdd, isEven. Use 
command line argument to pass a value to the object and perform the 
above operations.



class MyNumber {

    private int num;

    // Default constructor
    MyNumber() {
        num = 0;
    }

    // Parameterized constructor
    MyNumber(int n) {
        num = n;
    }

    void isNegative() {
        if (num < 0)
            System.out.println("Number is Negative");
        else
            System.out.println("Number is Not Negative");
    }

    void isPositive() {
        if (num > 0)
            System.out.println("Number is Positive");
        else
            System.out.println("Number is Not Positive");
    }

    void isOdd() {
        if (num % 2 != 0)
            System.out.println("Number is Odd");
        else
            System.out.println("Number is Not Odd");
    }

    void isEven() {
        if (num % 2 == 0)
            System.out.println("Number is Even");
        else
            System.out.println("Number is Not Even");
    }
}

public class Main {
    public static void main(String[] args) {

        int n = Integer.parseInt(args[0]);

        MyNumber obj = new MyNumber(n);

        System.out.println("Number = " + n);

        obj.isNegative();
        obj.isPositive();
        obj.isOdd();
        obj.isEven();
    }
}




Q2) Create a Node.js mini file-management application using the fs and path 
modules that allows the user to create, read, update, rename and delete 
text files through a menu-driven interface.



const fs = require('fs');
const path = require('path');
const readline = require('readline');

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

function menu() {
    console.log("\n--- File Management ---");
    console.log("1. Create File");
    console.log("2. Read File");
    console.log("3. Update File");
    console.log("4. Rename File");
    console.log("5. Delete File");
    console.log("6. Exit");

    rl.question("Enter your choice: ", choice => {

        switch (choice) {

            case '1':
                rl.question("Enter file name: ", file => {
                    rl.question("Enter content: ", data => {

                        fs.writeFile(path.join(__dirname, file),
                            data, err => {

                            if (err)
                                console.log("Error:", err);
                            else
                                console.log("File created successfully.");

                            menu();
                        });
                    });
                });
                break;

            case '2':
                rl.question("Enter file name: ", file => {

                    fs.readFile(path.join(__dirname, file),
                        'utf8', (err, data) => {

                        if (err)
                            console.log("Error:", err);
                        else
                            console.log("File Content:", data);

                        menu();
                    });
                });
                break;

            case '3':
                rl.question("Enter file name: ", file => {
                    rl.question("Enter new content: ", data => {

                        fs.writeFile(path.join(__dirname, file),
                            data, err => {

                            if (err)
                                console.log("Error:", err);
                            else
                                console.log("File updated successfully.");

                            menu();
                        });
                    });
                });
                break;

            case '4':
                rl.question("Enter old file name: ", oldFile => {
                    rl.question("Enter new file name: ", newFile => {

                        fs.rename(
                            path.join(__dirname, oldFile),
                            path.join(__dirname, newFile),
                            err => {

                            if (err)
                                console.log("Error:", err);
                            else
                                console.log("File renamed successfully.");

                            menu();
                        });
                    });
                });
                break;

            case '5':
                rl.question("Enter file name: ", file => {

                    fs.unlink(path.join(__dirname, file), err => {

                        if (err)
                            console.log("Error:", err);
                        else
                            console.log("File deleted successfully.");

                        menu();
                    });
                });
                break;

            case '6':
                console.log("Program Ended.");
                rl.close();
                break;

            default:
                console.log("Invalid choice.");
                menu();
        }
    });
}

menu();
