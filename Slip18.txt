Q1) Create an abstract class Shape with methods area & volume. Derive a 
class Cylinder (radius, height). Calculate area and volume.


abstract class Shape {

    abstract void area();

    abstract void volume();
}

class Cylinder extends Shape {

    double radius, height;

    Cylinder(double radius, double height) {
        this.radius = radius;
        this.height = height;
    }

    void area() {
        double a = 2 * Math.PI * radius * (radius + height);
        System.out.println("Surface Area = " + a);
    }

    void volume() {
        double v = Math.PI * radius * radius * height;
        System.out.println("Volume = " + v);
    }
}

public class Main {
    public static void main(String[] args) {

        Cylinder c = new Cylinder(5, 10);

        c.area();
        c.volume();
    }
}




Q2) Write a Node.js program to demonstrate synchronous and asynchronous 
file operations and compare their execution.


const fs = require('fs');

console.log("Program Started");

// Synchronous Operation
console.log("\nSynchronous Operation:");

console.time("Sync");

fs.writeFileSync('sync.txt', 'Hello from Synchronous File');

let data1 = fs.readFileSync('sync.txt', 'utf8');

console.log("File Content:", data1);

console.timeEnd("Sync");


// Asynchronous Operation
console.log("\nAsynchronous Operation:");

console.time("Async");

fs.writeFile('async.txt', 'Hello from Asynchronous File', (err) => {

    if (err) {
        console.log(err);
        return;
    }

    fs.readFile('async.txt', 'utf8', (err, data2) => {

        if (err) {
            console.log(err);
            return;
        }

        console.log("File Content:", data2);
        console.timeEnd("Async");
    });
});

console.log("Program Ended");
