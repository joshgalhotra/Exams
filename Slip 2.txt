Q1) Write a Java Program to Display Armstrong Numbers Between range. 
Accept range from user.

import java.util.*;

class Armstrong
{
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);

        int start, end;

        System.out.print("Enter starting number: ");
        start = sc.nextInt();

        System.out.print("Enter ending number: ");
        end = sc.nextInt();

        System.out.println("Armstrong numbers are:");

        for(int n = start; n <= end; n++)
        {
            int num = n;
            int sum = 0;

            while(num > 0)
            {
                int r = num % 10;
                sum = sum + (r * r * r);
                num = num / 10;
            }

            if(sum == n)
            {
                System.out.print(n + " ");
            }
        }
    }
}
