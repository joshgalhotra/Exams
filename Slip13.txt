Q1) Creating an application with multiple buttons (e.g., Red, Green, Blue) 
that print their respective color or change the background color when 
clicked.

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class ColorButtons extends Application {

    public void start(Stage stage) {

        Button red = new Button("Red");
        Button green = new Button("Green");
        Button blue = new Button("Blue");

        red.setOnAction(e -> {
            System.out.println("Red");
        });

        green.setOnAction(e -> {
            System.out.println("Green");
        });

        blue.setOnAction(e -> {
            System.out.println("Blue");
        });

        HBox root = new HBox(10, red, green, blue);

        Scene scene = new Scene(root, 400, 200);

        stage.setTitle("Color Buttons");
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}



Q2) Create a Node.js HTTP server that sends an appropriate response to a
client request.


const http = require('http');

const server = http.createServer((req, res) => {

    res.writeHead(200, {'Content-Type': 'text/html'});

    res.write("<h1>Hello from Node.js Server</h1>");
    res.write("<p>Request received successfully!</p>");

    res.end();
});

server.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});
