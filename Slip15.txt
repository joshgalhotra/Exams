Q1) Write a program to create parent class College (cno, cname, caddr) and 
derived class Department (dno, dname) from College. Write a necessary 
methods to display College details.


class College {
    int cno;
    String cname, caddr;

    College(int cno, String cname, String caddr) {
        this.cno = cno;
        this.cname = cname;
        this.caddr = caddr;
    }

    void displayCollege() {
        System.out.println("College No: " + cno);
        System.out.println("College Name: " + cname);
        System.out.println("College Address: " + caddr);
    }
}

class Department extends College {
    int dno;
    String dname;

    Department(int cno, String cname, String caddr,
               int dno, String dname) {
        super(cno, cname, caddr);
        this.dno = dno;
        this.dname = dname;
    }

    void display() {
        displayCollege();
        System.out.println("Department No: " + dno);
        System.out.println("Department Name: " + dname);
    }
}

public class Main {
    public static void main(String[] args) {

        Department d = new Department(
            101, "KTHM College", "Nashik",
            1, "Computer Science"
        );

        d.display();
    }
}




Q2) Write a Node.js program using the path module to perform operations
such as joining, resolving and extracting file/directory information.


const path = require('path');

let filePath = 'C:/College/BCS/JavaScript/program.js';

// Joining paths
let joinedPath = path.join('C:', 'College', 'BCS', 'program.js');
console.log("Joined Path:", joinedPath);

// Resolving path
let resolvedPath = path.resolve('College', 'BCS', 'program.js');
console.log("Resolved Path:", resolvedPath);

// Directory name
console.log("Directory:", path.dirname(filePath));

// File name
console.log("File Name:", path.basename(filePath));

// File extension
console.log("Extension:", path.extname(filePath));

// File name without extension
console.log("Name without Extension:",
    path.basename(filePath, path.extname(filePath)));
