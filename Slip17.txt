Q1) Write a program for multilevel inheritance such that Country is inherited 
from Continent. State is inherited from Country. Display the place, State, 
Country and Continent.


class Continent {
    String continent;

    Continent(String continent) {
        this.continent = continent;
    }
}

class Country extends Continent {
    String country;

    Country(String continent, String country) {
        super(continent);
        this.country = country;
    }
}

class State extends Country {
    String state;

    State(String continent, String country, String state) {
        super(continent, country);
        this.state = state;
    }

    void display(String place) {
        System.out.println("Place     : " + place);
        System.out.println("State     : " + state);
        System.out.println("Country   : " + country);
        System.out.println("Continent : " + continent);
    }
}

public class Main {
    public static void main(String[] args) {

        State s = new State(
            "Asia",
            "India",
            "Maharashtra"
        );

        s.display("Nashik");
    }
}




Q2) Write a Node.js program to create, write, read and append data to a file 
using the fs module.


const fs = require('fs');

// Create and Write file
fs.writeFile('abc.txt', 'Hello Node.js', (err) => {

    if (err) {
        console.log(err);
        return;
    }

    console.log("File created and written successfully.");

    // Read file
    fs.readFile('abc.txt', 'utf8', (err, data) => {

        if (err) {
            console.log(err);
            return;
        }

        console.log("File Content:");
        console.log(data);

        // Append data
        fs.appendFile('abc.txt', '\nWelcome to JavaScript', (err) => {

            if (err) {
                console.log(err);
                return;
            }

            console.log("Data appended successfully.");

            // Read again after append
            fs.readFile('abc.txt', 'utf8', (err, data) => {

                if (err) {
                    console.log(err);
                    return;
                }

                console.log("Updated File Content:");
                console.log(data);
            });
        });
    });
});
